cmake_minimum_required(VERSION 3.20)
project(MidnightPulseStyle LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
find_package(KF6WindowSystem REQUIRED)
add_library(midnightpulse MODULE midnightpulse.cpp)
target_link_libraries(midnightpulse PRIVATE Qt6::Widgets KF6::WindowSystem)
set_target_properties(midnightpulse PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins/styles")
